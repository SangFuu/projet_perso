cmake_minimum_required(VERSION 3.16)
project(PetitProjetPendu C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(pendu main.c mycode.c)
if(MSVC)
  target_compile_options(pendu PRIVATE /W4)
else()
  target_compile_options(pendu PRIVATE -Wall -Wextra -Wpedantic)
endif()
